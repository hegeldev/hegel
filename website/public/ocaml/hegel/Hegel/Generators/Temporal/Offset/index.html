Module Temporal.Offset

UTC offsets.

type t =
  1. | Known of int
  2. | Unknown

A UTC offset: a known number of seconds east of UTC within ±max_seconds, or the unknown offset of RFC 3339's -00:00.

val max_seconds : int

The largest magnitude of a known offset in seconds, 25:59:59.

val to_string : t -> string

to_string offset formats offset: +05:30, -00:44:30 when seconds are nonzero, or -00:00 for Unknown.

val pp : Stdlib.Format.formatter -> t -> unit

pp formatter offset prints to_string offset.

val sexp_of_t : t -> Sexplib0.Sexp.t

sexp_of_t offset is to_string offset as an atom.

val equal : t -> t -> bool

equal a b returns whether a and b are the same offset.

val compare : t -> t -> int

compare a b orders known offsets from west to east, after Unknown.

val hegel_generator : (t, printable) generator

offsets (), every known offset.
