Module Temporal.Time

Times of day.

type t = {
  1. hour : int;
  2. minute : int;
  3. second : int;
  4. nanosecond : int;
}

A time of day from 00:00:00 to 23:59:59.999999999: hour in [0, 23], minute and second in [0, 59] (no leap seconds) and nanosecond in [0, 999999999].

val min_value : t

Midnight, 00:00:00.

val max_value : t

The last nanosecond of the day, 23:59:59.999999999.

val to_string : t -> string

to_string time formats time: 02:30:00, or 02:30:00.5 when the fractional part is nonzero.

val pp : Stdlib.Format.formatter -> t -> unit

pp formatter time prints to_string time.

val sexp_of_t : t -> Sexplib0.Sexp.t

sexp_of_t time is to_string time as an atom.

val equal : t -> t -> bool

equal a b returns whether a and b have the same fields.

val compare : t -> t -> int

compare a b orders valid times from midnight on.

val hegel_generator : (t, printable) generator

times (), every time of day.
