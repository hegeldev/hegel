Module Temporal.Date

Calendar dates.

type t = {
  1. year : int;
  2. month : int;
  3. day : int;
}

A proleptic Gregorian calendar date: year in [-999999, 999999], month in [1, 12] and day in [1, days in the month].

val min_value : t

The first date, -999999-01-01.

val max_value : t

The last date, +999999-12-31.

val to_string : t -> string

to_string date formats date: 2024-03-10, with a signed year of at least six digits outside 0000 to 9999.

val pp : Stdlib.Format.formatter -> t -> unit

pp formatter date prints to_string date.

val sexp_of_t : t -> Sexplib0.Sexp.t

sexp_of_t date is to_string date as an atom.

val equal : t -> t -> bool

equal a b returns whether a and b have the same fields.

val compare : t -> t -> int

compare a b orders valid dates chronologically.

val hegel_generator : (t, printable) generator

dates (), every date.
