159 // True if a CI environment is detected, by probing the variables the
160 // common CI providers set. Used as the default for
161 // Settings::derandomize so runs are reproducible under CI.
162 inline bool in_ci() {
163 // expected == nullptr means "any value present satisfies it".
164 struct CiVar {
165 const char* name;
166 const char* expected;
167 };
168 static constexpr CiVar ci_vars[] = {
169 {"CI", nullptr},
170 {"TF_BUILD", "true"},
171 {"BUILDKITE", "true"},
172 {"CIRCLECI", "true"},
173 {"CIRRUS_CI", "true"},
174 {"CODEBUILD_BUILD_ID", nullptr},
175 {"GITHUB_ACTIONS", "true"},
176 {"GITLAB_CI", nullptr},
177 {"HEROKU_TEST_RUN_ID", nullptr},
178 {"TEAMCITY_VERSION", nullptr},
179 };
180 for (const CiVar& v : ci_vars) {
181 const char* value = std::getenv(v.name);
182 if (value == nullptr) {
183 continue;
184 }
185 if (v.expected == nullptr ||
186 std::strcmp(value, v.expected) == 0) {
187 return true;
188 }
189 }
190 return false;
191 }
192 } // namespace internal
