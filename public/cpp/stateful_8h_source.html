787 template <typename M>
788 void run_concurrent(M& machine, TestCase& tc, int64_t min_concurrency,
789 int64_t max_concurrency) {
790 static_assert(std::is_base_of<ConcurrentStateMachine<M>, M>::value,
791 "run_concurrent() requires a machine deriving from "
792 "ConcurrentStateMachine<M>.");
793 if (min_concurrency < 1 || min_concurrency > max_concurrency) {
794 throw std::invalid_argument(
795 "concurrency bounds must satisfy 1 <= min <= max");
796 }
797
798 std::vector<ConcurrentRule<M>> rules = machine.rules();
799 std::vector<Invariant<M>> invariants = machine.invariants();
800 if (rules.empty()) {
801 throw std::invalid_argument(
802 "Cannot run a concurrent state machine with no rules.");
803 }
804
805 std::vector<std::string> rule_names;
806 std::vector<std::string> invariant_names;
807 std::vector<std::string> group_names; // maps group ID to name
808 std::map<std::string, int64_t> group_ids; // maps group name to ID
809 std::vector<int64_t>
810 rule_groups; // ID of group that rule_names[i] belongs to is the ID
811 // at rule_groups[i].
812 rule_names.reserve(rules.size());
813 rule_groups.reserve(rules.size());
814 for (const ConcurrentRule<M>& rule : rules) {
815 rule_names.push_back(rule.name());
816 auto [group, inserted] = group_ids.emplace(
817 rule.group(), static_cast<int64_t>(group_ids.size()));
818 if (inserted) {
819 group_names.push_back(rule.group());
820 }
821 rule_groups.push_back(group->second);
822 }
823 invariant_names.reserve(invariants.size());
824 for (const Invariant<M>& invariant : invariants) {
825 invariant_names.push_back(invariant.name());
826 }
827
828 internal::StateMachineHandle machine_handle(
829 tc, rule_names, rule_groups, invariant_names, min_concurrency,
830 max_concurrency);
831 tc.note("Concurrency level: " +
832 std::to_string(machine_handle.concurrency()));
833 tc.note("Initial invariant check.");
834 check_invariants(tc, "in the initial state", machine, invariants);

839 std::optional<internal::CapturedException> exception;

841 struct RoundState {
842 std::mutex mutex;
843 std::condition_variable start;
844 std::condition_variable done;
845 uint64_t round_number = 0;
846 bool stop = false;
847 std::size_t num_workers_completed = 0;
848 std::vector<Event> events;
849 };

851 auto classify = [] {
852 Event event;
853 event.exception = internal::capture_current_exception();
854 try {
855 std::rethrow_exception(event.exception->exception);
856 } catch (const internal::HegelStopTest&) {
857 event.kind = EventKind::Overrun;
858 } catch (const internal::HegelReject&) {
859 // not the same as rule rejection - entire test case is rejected
860 // requires some worker race on cloned test cases with same
861 // parent
862 event.kind = EventKind::Invalid; // GCOVR_EXCL_LINE
863 } catch (const std::invalid_argument&) {
864 event.kind = EventKind::Control;
865 } catch (...) {
866 event.kind = EventKind::Panicked;
867 }
868 return event;
869 };

871 auto run_round = [&](std::size_t worker, TestCase& worker_tc) {
872 try {
873 while (true) {
874 int64_t rule_index = machine_handle.next_rule(
875 worker_tc, static_cast<int64_t>(worker));
876 if (rule_index == internal::state_machine_done) {
877 return Event{};
878 }
879 if (rule_index < 0 ||
880 static_cast<std::size_t>(rule_index) >= rules.size()) {
881 // GCOVR_EXCL_START
882 throw std::runtime_error(
883 "state_machine_next_rule returned out-of-range "
884 "rule index. Please report this as a bug.");
885 // GCOVR_EXCL_STOP
886 }
887
888 const ConcurrentRule<M>& rule =
889 rules[static_cast<std::size_t>(rule_index)];
890 worker_tc.note("Rule: " + rule.name());
891 try {
892 internal::NoteIndentScope indent(worker_tc);
893 rule.function()(worker_tc, machine);
894 } catch (const internal::HegelReject&) {
895 machine_handle.rule_rejected(
896 worker_tc, static_cast<int64_t>(worker));
897 worker_tc.note(
898 "Rule stopped early due to violated assumption.");
899 }
900 }
901 } catch (...) {
902 return classify();
903 }
904 };
905
906 std::size_t worker_count =
907 static_cast<std::size_t>(machine_handle.concurrency());
908 RoundState state;
909 state.events.resize(worker_count);
910 std::vector<std::thread> workers;
911 workers.reserve(worker_count);
912 for (std::size_t worker = 0; worker < worker_count; ++worker) {
913 TestCase worker_tc = tc.clone();
914 internal::set_concurrent_worker(worker_tc, worker);
915 workers.emplace_back(
916 [&, worker, worker_tc = std::move(worker_tc)]() mutable {
917 uint64_t observed_round_number = 0;
918 while (true) {
919 {
920 std::unique_lock<std::mutex> lock(state.mutex);
921 state.start.wait(lock, [&] {
922 return state.stop ||
923 state.round_number !=
924 observed_round_number; // main thread
925 // started a
926 // new round
927 });
928 if (state.stop) {
929 return;
930 }
931 observed_round_number = state.round_number;
932 }
933
934 {
935 Event event = run_round(worker, worker_tc);
936 std::lock_guard<std::mutex> lock(state.mutex);
937 state.events[worker] = std::move(event);
938 ++state.num_workers_completed;
939 }
940 state.done.notify_one();
941 }
942 });
943 }
944
945 auto stop_workers = [&] {
946 {
947 std::lock_guard<std::mutex> lock(state.mutex);
948 state.stop = true;
949 }
950 state.start.notify_all();
951 for (std::thread& worker : workers) {
952 worker.join();
953 }
954 };
955
956 try {
957 while (true) {
958 int64_t group = machine_handle.next_group(tc);
959 if (group == internal::state_machine_done) {
960 break;
961 }
962 if (group < 0 ||
963 static_cast<std::size_t>(group) >= group_names.size()) {
964 // GCOVR_EXCL_START
965 throw std::runtime_error(
966 "state_machine_next_group returned an unknown group "
967 "index. Please report this as a bug.");
968 // GCOVR_EXCL_STOP
969 }
970 {
971 std::lock_guard<std::mutex> lock(state.mutex);
972 state.num_workers_completed = 0;
973 ++state.round_number;
974 tc.note("---------------- Round " +
975 std::to_string(state.round_number) + ": group \"" +
976 group_names[static_cast<std::size_t>(group)] +
977 "\" ----------------");
978 }
979 state.start.notify_all();
980 {
981 std::unique_lock<std::mutex> lock(state.mutex);
982 state.done.wait(lock, [&] {
983 return state.num_workers_completed == worker_count;
984 });
985 }
986
987 std::optional<std::size_t> control;
988 bool saw_overrun = false;
989 bool saw_invalid = false;
990 std::optional<std::size_t> panic;
991 // need to collect every event because the precedence is
992 // control > overrun > invalid > panic > round complete
993 for (std::size_t worker = 0; worker < worker_count; ++worker) {
994 switch (state.events[worker].kind) {
995 case EventKind::RoundDone:
996 break;
997 // GCOVR_EXCL_START
998 case EventKind::Invalid:
999 saw_invalid = true;
1000 break;
1001 // GCOVR_EXCL_STOP
1002 case EventKind::Control:
1003 if (!control.has_value()) {
1004 control = worker;
1005 }
1006 break;
1007 case EventKind::Overrun:
1008 saw_overrun = true;
1009 break;
1010 case EventKind::Panicked:
1011 if (!panic.has_value()) {
1012 panic = worker;
1013 }
1014 break;
1015 }
1016 }
1017 if (control.has_value()) {
1018 state.events[*control].exception->rethrow();
1019 }
1020 if (saw_overrun) {
1021 throw internal::HegelStopTest();
1022 }
1023 if (saw_invalid) {
1024 throw internal::HegelReject(); // GCOVR_EXCL_LINE
1025 }
1026 if (panic.has_value()) {
1027 state.events[*panic].exception->rethrow();
1028 }
1029 check_invariants(
1030 tc, "after round " + std::to_string(state.round_number),
1031 machine, invariants, std::ref(machine_handle));
1032 }
1033
1034 tc.note("Final invariant check.");
1035 check_invariants(tc, "in the final state", machine, invariants);
1036 } catch (...) {
1037 stop_workers();
1038 throw;
1039 }
1040 stop_workers();
1041 }

1067 template <typename M>
1068 void run(M& machine, TestCase& tc, const RunParams& params = {}) {
1069 static_assert(
1070 std::is_base_of<StateMachine<M, typename M::state_type>, M>::value,
1071 "run() requires a machine deriving from "
1072 "StateMachine<M, State>.");
1073 std::vector<Rule<M>> rules = machine.rules();
1074 std::vector<Invariant<M>> invariants = machine.invariants();
1075 if (rules.empty()) {
1076 throw std::invalid_argument(
1077 "Cannot run a state machine with no rules.");
1078 }
1079 std::vector<std::string> rule_names;
1080 rule_names.reserve(rules.size());
1081 for (const Rule<M>& rule : rules)
1082 rule_names.push_back(rule.name());
1083
1084 std::vector<std::string> invariant_names;
1085 invariant_names.reserve(invariants.size());
1086 for (const Invariant<M>& invariant : invariants)
1087 invariant_names.push_back(invariant.name());
1088
1089 std::vector<int64_t> rule_groups(rule_names.size(), 0);
1090 internal::StateMachineHandle machine_handle(tc, rule_names, rule_groups,
1091 invariant_names, 1, 1);
1092 tc.note("Initial invariant check.");
1093 print_state(tc, machine, params);
1094 check_invariants(tc, "in the initial state", machine, invariants);
1095 int64_t steps_run = 0;
1096
1097 // Sequential stateful testing is modeled as a concurrent stateful test
1098 // with one group and one worker.
1099 while (machine_handle.next_group(tc) != internal::state_machine_done) {
1100 while (true) {
1101 int64_t next_rule_idx = machine_handle.next_rule(tc, 0);
1102 if (next_rule_idx == internal::state_machine_done) {
1103 break;
1104 }
1105 // GCOVR_EXCL_START
1106 if (next_rule_idx < 0 ||
1107 static_cast<size_t>(next_rule_idx) >= rules.size()) {
1108 throw std::runtime_error(
1109 "state_machine_next_rule returned out-of-range "
1110 "rule index. Please report this as a bug.");
1111 }
1112 // GCOVR_EXCL_STOP
1113 internal::start_span(tc, internal::SpanLabel::StatefulRule);
1114 steps_run++;
1115 const Rule<M>& rule = rules[static_cast<size_t>(next_rule_idx)];
1116 tc.note("Step " + std::to_string(steps_run) + ": " +
1117 rule.name());
1118
1119 try {
1120 // nest the draws the step makes under its "Step N" header.
1121 {
1122 internal::NoteIndentScope indent(tc);
1123 rule.step()(tc, machine);
1124 }
1125 print_state(tc, machine, params);
1126 internal::stop_span(tc);
1127 } catch (const internal::HegelReject&) {
1128 tc.note("Rule stopped early due to violated assumption.");
1129 machine_handle.rule_rejected(tc, 0);
1130 internal::stop_span(tc, true);
1131 } catch (...) {
1132 internal::stop_span(tc);
1133 throw;
1134 }
1135 }
1136 check_invariants(tc, "after step " + std::to_string(steps_run),
1137 machine, invariants, std::ref(machine_handle));
1138 }
1139
1140 tc.note("Final invariant check.");
1141 check_invariants(tc, "in the final state", machine, invariants);
1142 }
