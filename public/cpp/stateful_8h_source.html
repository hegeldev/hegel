191 hegel::internal::draw_variable(tc, p_.pool_id_, consume_);
192
193 auto it = p_.pool_.find(variable);
194 // GCOVR_EXCL_START
195 if (it == p_.pool_.end()) {
196 throw std::runtime_error(
197 "Pool state diverged between the engine and the "
198 "client, or a bug in the pool bookkeeping.");
199 }
200 // GCOVR_EXCL_STOP
201 if (consume_) {
202 auto val = std::move(it->second);
203 p_.pool_.erase(it);
204 return val;
205 }
206 return it->second;
207 }

209 private:
210 Pool<T>& p_;
211 bool consume_;

509 template <typename M>
510 void run(M& machine, TestCase& tc, const RunParams& params = {}) {
511 static_assert(
512 std::is_base_of<StateMachine<M, typename M::state_type>, M>::value,
513 "run() requires a machine deriving from "
514 "StateMachine<M, State>.");
515 std::vector<Rule<M>> rules = machine.rules();
516 std::vector<Invariant<M>> invariants = machine.invariants();
517 if (rules.empty()) {
518 throw std::invalid_argument(
519 "Cannot run a state machine with no rules.");
520 }
521 bool is_single = internal::is_single_test_case(tc);
522 int64_t max_steps = is_single ? std::numeric_limits<int64_t>::max()
523 : internal::stateful_step_count(tc);
524
525 std::vector<std::string> rule_names;
526 rule_names.reserve(rules.size());
527 for (const Rule<M>& rule : rules)
528 rule_names.push_back(rule.name());
529
530 std::vector<std::string> invariant_names;
531 invariant_names.reserve(invariants.size());
532 for (const Invariant<M>& invariant : invariants)
533 invariant_names.push_back(invariant.name());
534
535 print_state(tc, machine, params);
536 check_invariants(tc, "in the initial state", machine, invariants);
537
538 auto must_stop = [=](int64_t steps_run) -> std::optional<bool> {
539 if (is_single) {
540 return false;
541 }
542 if (steps_run >= max_steps) {
543 return true;
544 }
545 if (steps_run <= 0) {
546 return false;
547 }
548 return std::nullopt;
549 };
550 int64_t state_machine_id =
551 internal::new_state_machine(tc, rule_names, invariant_names);
552 int64_t steps_run = 0;
553 int64_t num_steps_succeeded = 0;
554 constexpr double p_stop = 1.0 / 65536.0; // 2^-16
555
556 while (true) {
557 internal::start_span(tc, internal::SpanLabel::StatefulRule);
558 // gives engine more control of when to stop generating steps
559 if (internal::draw_boolean(tc, p_stop, must_stop(steps_run))) {
560 internal::stop_span(tc);
561 if (num_steps_succeeded == 0) {
562 tc.reject();
563 }
564 break;
565 } else {
566 steps_run++;
567 try {
568 int64_t next_rule_idx =
569 internal::draw_rule(tc, state_machine_id);
570 // GCOVR_EXCL_START
571 if (next_rule_idx < 0 ||
572 static_cast<size_t>(next_rule_idx) >= rules.size()) {
573 throw std::runtime_error(
574 "state_machine_next_rule returned out-of-range "
575 "rule index. Please report this as a bug.");
576 }
577 // GCOVR_EXCL_STOP
578 const Rule<M>& rule =
579 rules[static_cast<size_t>(next_rule_idx)];
580 tc.note("Step " + std::to_string(steps_run) + ": " +
581 rule.name());
582
583 // nest the draws the step makes under its "Step N" header.
584 {
585 internal::NoteIndentScope indent(tc);
586 rule.step()(tc, machine);
587 }
588 print_state(tc, machine, params);
589 check_invariants(tc,
590 "after step " + std::to_string(steps_run),
591 machine, invariants);
592 internal::stop_span(tc);
593 num_steps_succeeded++;
594 } catch (const internal::HegelReject&) {
595 tc.note("Rule stopped early due to violated assumption.");
596 internal::stop_span(tc, true);
597 } catch (...) {
598 internal::stop_span(tc);
599 throw;
600 }
601 }
602 }
603 }
