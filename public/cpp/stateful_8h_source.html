179 // Concrete IGenerator for variables.
180 template <typename T> class VariablesGenerator : public IGenerator<T> {
181 public:
182 explicit VariablesGenerator(Pool<T>& p, bool consume)
183 : p_(p), consume_(consume) {}
184
185 T do_draw(const TestCase& tc) const override {
186 int64_t variable =
187 hegel::internal::draw_variable(tc, p_.pool_id_, consume_);
188
189 auto it = p_.pool_.find(variable);
190 // GCOVR_EXCL_START
191 if (it == p_.pool_.end()) {
192 throw std::runtime_error(
193 "Pool state diverged between the engine and the "
194 "client, or a bug in the pool bookkeeping.");
195 }
196 // GCOVR_EXCL_STOP
197 if (consume_) {
198 auto val = std::move(it->second);
199 p_.pool_.erase(it);
200 return val;
201 }
202 return it->second;
203 }
204
205 private:
206 Pool<T>& p_;
207 bool consume_;
208 };

429 template <typename M> void run(M& machine, TestCase& tc) {
430 static_assert(
431 std::is_base_of<StateMachine<M>, M>::value,
432 "run() requires a machine deriving from StateMachine<M>.");
433 std::vector<Rule<M>> rules = machine.rules();
434 std::vector<Invariant<M>> invariants = machine.invariants();
435 if (rules.empty()) {
436 throw std::invalid_argument(
437 "Cannot run a state machine with no rules.");
438 }
439 bool is_single = internal::is_single_test_case(tc);
440 int64_t max_steps = is_single ? std::numeric_limits<int64_t>::max()
441 : internal::stateful_step_count(tc);
442
443 std::vector<std::string> rule_names;
444 rule_names.reserve(rules.size());
445 for (const Rule<M>& rule : rules)
446 rule_names.push_back(rule.name());
447
448 std::vector<std::string> invariant_names;
449 invariant_names.reserve(invariants.size());
450 for (const Invariant<M>& invariant : invariants)
451 invariant_names.push_back(invariant.name());
452
453 check_invariants(tc, "in the initial state", machine, invariants);
454
455 auto must_stop = [=](int64_t steps_run) -> std::optional<bool> {
456 if (is_single) {
457 return false;
458 }
459 if (steps_run >= max_steps) {
460 return true;
461 }
462 if (steps_run <= 0) {
463 return false;
464 }
465 return std::nullopt;
466 };
467 int64_t state_machine_id =
468 internal::new_state_machine(tc, rule_names, invariant_names);
469 int64_t steps_run = 0;
470 int64_t num_steps_succeeded = 0;
471 constexpr double p_stop = 1.0 / 65536.0; // 2^-16
472
473 while (true) {
474 internal::start_span(tc, internal::SpanLabel::StatefulRule);
475 // gives engine more control of when to stop generating steps
476 if (internal::draw_boolean(tc, p_stop, must_stop(steps_run),
477 /*silent=*/true)) {
478 internal::stop_span(tc);
479 if (num_steps_succeeded == 0) {
480 tc.reject();
481 }
482 break;
483 } else {
484 steps_run++;
485 try {
486 int64_t next_rule_idx =
487 internal::draw_rule(tc, state_machine_id);
488 // GCOVR_EXCL_START
489 if (next_rule_idx < 0 ||
490 static_cast<size_t>(next_rule_idx) >= rules.size()) {
491 throw std::runtime_error(
492 "state_machine_next_rule returned out-of-range "
493 "rule index. Please report this as a bug.");
494 }
495 // GCOVR_EXCL_STOP
496 const Rule<M>& rule =
497 rules[static_cast<size_t>(next_rule_idx)];
498 tc.note("Step " + std::to_string(steps_run) + ": " +
499 rule.name());
500
501 // nest the draws the step makes under its "Step N" header.
502 {
503 internal::NoteIndentScope indent(tc);
504 rule.step()(tc, machine);
505 }
506 check_invariants(tc,
507 "after step " + std::to_string(steps_run),
508 machine, invariants);
509 internal::stop_span(tc);
510 num_steps_succeeded++;
511 } catch (const internal::HegelReject&) {
512 tc.note("Rule stopped early due to violated assumption.");
513 internal::stop_span(tc, true);
514 } catch (...) {
515 internal::stop_span(tc);
516 throw;
517 }
518 }
519 }
520 }
