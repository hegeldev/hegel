186 : p_(p), consume_(consume) {}

188 T do_draw(const TestCase& tc) const override {
189 int64_t variable =
190 hegel::internal::draw_variable(tc, p_.pool_id_, consume_);
191
192 auto it = p_.pool_.find(variable);
193 // GCOVR_EXCL_START
194 if (it == p_.pool_.end()) {
195 throw std::runtime_error(
196 "Pool state diverged between the engine and the "
197 "client, or a bug in the pool bookkeeping.");
198 }
199 // GCOVR_EXCL_STOP
200 if (consume_) {
201 auto val = std::move(it->second);
202 p_.pool_.erase(it);
203 return val;
204 }
205 return it->second;
206 }

208 private:
209 Pool<T>& p_;
210 bool consume_;
211 };

510 template <typename M>
511 void run(M& machine, TestCase& tc, const RunParams& params = {}) {
512 static_assert(
513 std::is_base_of<StateMachine<M, typename M::state_type>, M>::value,
514 "run() requires a machine deriving from "
515 "StateMachine<M, State>.");
516 std::vector<Rule<M>> rules = machine.rules();
517 std::vector<Invariant<M>> invariants = machine.invariants();
518 if (rules.empty()) {
519 throw std::invalid_argument(
520 "Cannot run a state machine with no rules.");
521 }
522 std::vector<std::string> rule_names;
523 rule_names.reserve(rules.size());
524 for (const Rule<M>& rule : rules)
525 rule_names.push_back(rule.name());
526
527 std::vector<std::string> invariant_names;
528 invariant_names.reserve(invariants.size());
529 for (const Invariant<M>& invariant : invariants)
530 invariant_names.push_back(invariant.name());
531
532 print_state(tc, machine, params);
533 check_invariants(tc, "in the initial state", machine, invariants);
534
535 int64_t state_machine_id =
536 internal::new_state_machine(tc, rule_names, invariant_names);
537 int64_t steps_run = 0;
538
539 while (true) {
540 internal::start_span(tc, internal::SpanLabel::StatefulRule);
541 int64_t next_rule_idx = internal::draw_rule(tc, state_machine_id);
542 if (next_rule_idx == internal::state_machine_done) {
543 break;
544 }
545 // GCOVR_EXCL_START
546 if (next_rule_idx < 0 ||
547 static_cast<size_t>(next_rule_idx) >= rules.size()) {
548 throw std::runtime_error(
549 "state_machine_next_rule returned out-of-range "
550 "rule index. Please report this as a bug.");
551 }
552 // GCOVR_EXCL_STOP
553 steps_run++;
554 const Rule<M>& rule = rules[static_cast<size_t>(next_rule_idx)];
555 tc.note("Step " + std::to_string(steps_run) + ": " + rule.name());
556
557 try {
558 // nest the draws the step makes under its "Step N" header.
559 {
560 internal::NoteIndentScope indent(tc);
561 rule.step()(tc, machine);
562 }
563 print_state(tc, machine, params);
564 check_invariants(tc, "after step " + std::to_string(steps_run),
565 machine, invariants);
566 internal::stop_span(tc);
567 } catch (const internal::HegelReject&) {
568 tc.note("Rule stopped early due to violated assumption.");
569 internal::stop_span(tc, true);
570 } catch (...) {
571 internal::stop_span(tc);
572 throw;
573 }
574 }
575 }
