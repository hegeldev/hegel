854 template <typename M>
855 void run_concurrent(M& machine, TestCase& tc, int64_t min_concurrency,
856 int64_t max_concurrency) {
857 static_assert(std::is_base_of<ConcurrentStateMachine<M>, M>::value,
858 "run_concurrent() requires a machine deriving from "
859 "ConcurrentStateMachine<M>.");
860 if (min_concurrency < 1 || min_concurrency > max_concurrency) {
861 throw std::invalid_argument(
862 "concurrency bounds must satisfy 1 <= min <= max");
863 }
864
865 std::vector<ConcurrentRule<M>> rules = machine.rules();
866 std::vector<Invariant<M>> invariants = machine.invariants();
867 if (rules.empty()) {
868 throw std::invalid_argument(
869 "Cannot run a concurrent state machine with no rules.");
870 }
871
872 std::vector<std::string> rule_names;
873 std::vector<std::string> invariant_names;
874 std::vector<std::string> group_names; // maps group ID to name
875 std::map<std::string, int64_t> group_ids; // maps group name to ID
876 std::vector<int64_t>
877 rule_groups; // ID of group that rule_names[i] belongs to is the ID
878 // at rule_groups[i].
879 rule_names.reserve(rules.size());
880 rule_groups.reserve(rules.size());
881 for (const ConcurrentRule<M>& rule : rules) {
882 rule_names.push_back(rule.name());
883 auto [group, inserted] = group_ids.emplace(
884 rule.group(), static_cast<int64_t>(group_ids.size()));
885 if (inserted) {
886 group_names.push_back(rule.group());
887 }
888 rule_groups.push_back(group->second);
889 }
890 invariant_names.reserve(invariants.size());
891 for (const Invariant<M>& invariant : invariants) {
892 invariant_names.push_back(invariant.name());
893 }
894
895 internal::StateMachineHandle machine_handle(
896 tc, rule_names, rule_groups, invariant_names, min_concurrency,
897 max_concurrency);
898 tc.note("Concurrency level: " +
899 std::to_string(machine_handle.concurrency()));
900 tc.note("Initial invariant check.");
901 check_invariants(tc, "in the initial state", machine, invariants);

905 EventKind kind = EventKind::RoundDone;
906 std::optional<internal::CapturedException> exception;

908 struct RoundState {
909 std::mutex mutex;
910 std::condition_variable start;
911 std::condition_variable done;
912 uint64_t round_number = 0;
913 bool stop = false;
914 std::size_t num_workers_completed = 0;
915 std::vector<Event> events;
916 };

918 auto classify = [] {
919 Event event;
920 event.exception = internal::capture_current_exception();
921 try {
922 std::rethrow_exception(event.exception->exception);
923 } catch (const internal::HegelStopTest&) {
924 event.kind = EventKind::Overrun;
925 } catch (const internal::HegelReject&) {
926 // not the same as rule rejection - entire test case is rejected
927 // requires some worker race on cloned test cases with same
928 // parent
929 event.kind = EventKind::Invalid; // GCOVR_EXCL_LINE
930 } catch (const std::invalid_argument&) {
931 event.kind = EventKind::Control;
932 } catch (...) {
933 event.kind = EventKind::Panicked;
934 }
935 return event;
936 };

938 auto run_round = [&](std::size_t worker, TestCase& worker_tc) {
939 try {
940 while (true) {
941 int64_t rule_index = machine_handle.next_rule(
942 worker_tc, static_cast<int64_t>(worker));
943 if (rule_index == internal::state_machine_done) {
944 return Event{};
945 }
946 if (rule_index < 0 ||
947 static_cast<std::size_t>(rule_index) >= rules.size()) {
948 // GCOVR_EXCL_START
949 throw std::runtime_error(
950 "state_machine_next_rule returned out-of-range "
951 "rule index. Please report this as a bug.");
952 // GCOVR_EXCL_STOP
953 }
954
955 const ConcurrentRule<M>& rule =
956 rules[static_cast<std::size_t>(rule_index)];
957 worker_tc.note("Rule: " + rule.name());
958 try {
959 internal::NoteIndentScope indent(worker_tc);
960 rule.function()(worker_tc, machine);
961 } catch (const internal::HegelReject&) {
962 machine_handle.rule_rejected(
963 worker_tc, static_cast<int64_t>(worker));
964 worker_tc.note(
965 "Rule stopped early due to violated assumption.");
966 }
967 }
968 } catch (...) {
969 return classify();
970 }
971 };
972
973 std::size_t worker_count =
974 static_cast<std::size_t>(machine_handle.concurrency());
975 RoundState state;
976 state.events.resize(worker_count);
977 std::vector<std::thread> workers;
978 workers.reserve(worker_count);
979 for (std::size_t worker = 0; worker < worker_count; ++worker) {
980 TestCase worker_tc = tc.clone();
981 internal::set_concurrent_worker(worker_tc, worker);
982 workers.emplace_back(
983 [&, worker, worker_tc = std::move(worker_tc)]() mutable {
984 uint64_t observed_round_number = 0;
985 while (true) {
986 {
987 std::unique_lock<std::mutex> lock(state.mutex);
988 state.start.wait(lock, [&] {
989 return state.stop ||
990 state.round_number !=
991 observed_round_number; // main thread
992 // started a
993 // new round
994 });
995 if (state.stop) {
996 return;
997 }
998 observed_round_number = state.round_number;
999 }
1000
1001 {
1002 Event event = run_round(worker, worker_tc);
1003 std::lock_guard<std::mutex> lock(state.mutex);
1004 state.events[worker] = std::move(event);
1005 ++state.num_workers_completed;
1006 }
1007 state.done.notify_one();
1008 }
1009 });
1010 }
1011
1012 auto stop_workers = [&] {
1013 {
1014 std::lock_guard<std::mutex> lock(state.mutex);
1015 state.stop = true;
1016 }
1017 state.start.notify_all();
1018 for (std::thread& worker : workers) {
1019 worker.join();
1020 }
1021 };
1022
1023 try {
1024 while (true) {
1025 int64_t group = machine_handle.next_group(tc);
1026 if (group == internal::state_machine_done) {
1027 break;
1028 }
1029 if (group < 0 ||
1030 static_cast<std::size_t>(group) >= group_names.size()) {
1031 // GCOVR_EXCL_START
1032 throw std::runtime_error(
1033 "state_machine_next_group returned an unknown group "
1034 "index. Please report this as a bug.");
1035 // GCOVR_EXCL_STOP
1036 }
1037 {
1038 std::lock_guard<std::mutex> lock(state.mutex);
1039 state.num_workers_completed = 0;
1040 ++state.round_number;
1041 tc.note("---------------- Round " +
1042 std::to_string(state.round_number) + ": group \"" +
1043 group_names[static_cast<std::size_t>(group)] +
1044 "\" ----------------");
1045 }
1046 state.start.notify_all();
1047 {
1048 std::unique_lock<std::mutex> lock(state.mutex);
1049 state.done.wait(lock, [&] {
1050 return state.num_workers_completed == worker_count;
1051 });
1052 }
1053
1054 std::optional<std::size_t> control;
1055 bool saw_overrun = false;
1056 bool saw_invalid = false;
1057 std::optional<std::size_t> panic;
1058 // need to collect every event because the precedence is
1059 // control > overrun > invalid > panic > round complete
1060 for (std::size_t worker = 0; worker < worker_count; ++worker) {
1061 switch (state.events[worker].kind) {
1062 case EventKind::RoundDone:
1063 break;
1064 // GCOVR_EXCL_START
1065 case EventKind::Invalid:
1066 saw_invalid = true;
1067 break;
1068 // GCOVR_EXCL_STOP
1069 case EventKind::Control:
1070 if (!control.has_value()) {
1071 control = worker;
1072 }
1073 break;
1074 case EventKind::Overrun:
1075 saw_overrun = true;
1076 break;
1077 case EventKind::Panicked:
1078 if (!panic.has_value()) {
1079 panic = worker;
1080 }
1081 break;
1082 }
1083 }
1084 if (control.has_value()) {
1085 state.events[*control].exception->rethrow();
1086 }
1087 if (saw_overrun) {
1088 throw internal::HegelStopTest();
1089 }
1090 if (saw_invalid) {
1091 throw internal::HegelReject(); // GCOVR_EXCL_LINE
1092 }
1093 if (panic.has_value()) {
1094 state.events[*panic].exception->rethrow();
1095 }
1096 check_invariants(
1097 tc, "after round " + std::to_string(state.round_number),
1098 machine, invariants, std::ref(machine_handle));
1099 }
1100
1101 tc.note("Final invariant check.");
1102 check_invariants(tc, "in the final state", machine, invariants);
1103 } catch (...) {
1104 stop_workers();
1105 throw;
1106 }
1107 stop_workers();
1108 }

1134 template <typename M>
1135 void run(M& machine, TestCase& tc, const RunParams& params = {}) {
1136 static_assert(
1137 std::is_base_of<StateMachine<M, typename M::state_type>, M>::value,
1138 "run() requires a machine deriving from "
1139 "StateMachine<M, State>.");
1140 std::vector<Rule<M>> rules = machine.rules();
1141 std::vector<Invariant<M>> invariants = machine.invariants();
1142 if (rules.empty()) {
1143 throw std::invalid_argument(
1144 "Cannot run a state machine with no rules.");
1145 }
1146 std::vector<std::string> rule_names;
1147 rule_names.reserve(rules.size());
1148 for (const Rule<M>& rule : rules)
1149 rule_names.push_back(rule.name());
1150
1151 std::vector<std::string> invariant_names;
1152 invariant_names.reserve(invariants.size());
1153 for (const Invariant<M>& invariant : invariants)
1154 invariant_names.push_back(invariant.name());
1155
1156 std::vector<int64_t> rule_groups(rule_names.size(), 0);
1157 internal::StateMachineHandle machine_handle(tc, rule_names, rule_groups,
1158 invariant_names, 1, 1);
1159 tc.note("Initial invariant check.");
1160 print_state(tc, machine, params);
1161 check_invariants(tc, "in the initial state", machine, invariants);
1162 int64_t steps_run = 0;
1163
1164 // Sequential stateful testing is modeled as a concurrent stateful test
1165 // with one group and one worker.
1166 while (machine_handle.next_group(tc) != internal::state_machine_done) {
1167 while (true) {
1168 int64_t next_rule_idx = machine_handle.next_rule(tc, 0);
1169 if (next_rule_idx == internal::state_machine_done) {
1170 break;
1171 }
1172 // GCOVR_EXCL_START
1173 if (next_rule_idx < 0 ||
1174 static_cast<size_t>(next_rule_idx) >= rules.size()) {
1175 throw std::runtime_error(
1176 "state_machine_next_rule returned out-of-range "
1177 "rule index. Please report this as a bug.");
1178 }
1179 // GCOVR_EXCL_STOP
1180 internal::start_span(tc, internal::SpanLabel::StatefulRule);
1181 steps_run++;
1182 const Rule<M>& rule = rules[static_cast<size_t>(next_rule_idx)];
1183 tc.note("Step " + std::to_string(steps_run) + ": " +
1184 rule.name());
1185
1186 try {
1187 // nest the draws the step makes under its "Step N" header.
1188 {
1189 internal::NoteIndentScope indent(tc);
1190 rule.step()(tc, machine);
1191 }
1192 print_state(tc, machine, params);
1193 internal::stop_span(tc);
1194 } catch (const internal::HegelReject&) {
1195 tc.note("Rule stopped early due to violated assumption.");
1196 machine_handle.rule_rejected(tc, 0);
1197 internal::stop_span(tc, true);
1198 } catch (...) {
1199 internal::stop_span(tc);
1200 throw;
1201 }
1202 }
1203 check_invariants(tc, "after step " + std::to_string(steps_run),
1204 machine, invariants, std::ref(machine_handle));
1205 }
1206
1207 tc.note("Final invariant check.");
1208 check_invariants(tc, "in the final state", machine, invariants);
1209 }
