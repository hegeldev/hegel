574// Stringification takes two steps because `#` reads its argument before any
575// macro in it expands. `HEGEL_INTERNAL_STRINGIFY_(__LINE__)` alone therefore
576// gives "__LINE__". Passing the argument through a macro that does not apply
577// `#` expands it first, so the second macro stringifies the line number.
578#define HEGEL_INTERNAL_STRINGIFY_(x) #x
579#define HEGEL_INTERNAL_STRINGIFY(x) HEGEL_INTERNAL_STRINGIFY_(x)

581// The call site, which the engine uses to tell one requirement from another.
582// The three parts join into one string literal at compile time.
583//
584// `#` applies only to a parameter of a macro that takes parameters, so this
585// macro cannot stringify __LINE__ itself. A bare `#__LINE__` here leaves the
586// `#` in the token stream and fails to compile.
587#define HEGEL_INTERNAL_ORIGIN __FILE__ ":" HEGEL_INTERNAL_STRINGIFY(__LINE__)
