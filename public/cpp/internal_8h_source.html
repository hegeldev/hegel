129 /* Exception thrown when the backend tells us to abandon the current
130 * test iteration entirely (StopTest, Overflow, FlakyStrategyDefinition,

134 class HegelStopTest : public std::exception {

141 /* Exception thrown by a failed HEGEL_REQUIRE / HEGEL_REQUIRE_EQUAL. It
142 * carries the call site as its origin. The engine groups counterexamples
143 * by origin, so an origin taken from the throw site inside the library
144 * would make every failed requirement in a run look like one bug.
145 */
146 class HegelRequireFailure : public std::runtime_error {
147 public:
148 HegelRequireFailure(std::string origin, const std::string& message)
149 : std::runtime_error(message), origin_(std::move(origin)) {}
150
151 const std::string& origin() const noexcept { return origin_; }
152
153 private:
154 std::string origin_;
155 };

190 // Implementation of HEGEL_REQUIRE_EQUAL. The two values are compared as
191 // repr() renders them, so neither needs an operator==. That rendering is
192 // the comparison, so it happens on every call.
193 template <typename T>
194 void require_equal_impl(
195 const TestCase& tc, const char* origin, const T& left, const T& right,
196 std::string_view message = "HEGEL_REQUIRE_EQUAL: values differ") {
197 std::string left_repr = repr(left);
198 std::string right_repr = repr(right);
199 if (left_repr == right_repr) {
200 return;
201 }
202 std::string text(message);
203 // note() drops a message that will not print, but only after its
204 // argument is built. This guard keeps the two value trees and the
205 // descent through them out of every case that fails on the way to
206 // the counterexample, which is most of them.
207 if (diagnostics_visible(tc)) {
208 tc.note(text + " (- lhs / + rhs):\n" +
209 render_node_diff(repr_node(left), repr_node(right)));
210 }
211 fail_require(origin, text);
212 }
