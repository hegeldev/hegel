116 /* Exception thrown when a test case is rejected and should be
117 * discarded (e.g. by `TestCase::assume(false)`, an exhausted
118 * `filter()`, or an `UnsatisfiedAssumption` from the engine).
119 * The runner records the case as INVALID and continues.
120 */
121 class HegelReject : public std::exception {
122 public:
123 const char* what() const noexcept override {
124 return "test case rejected";
125 }
126 };

128 /* Exception thrown when the backend tells us to abandon the current
129 * test iteration entirely (StopTest, Overflow, FlakyStrategyDefinition,
130 * FlakyReplay). The runner unwinds the test body and skips
131 * `mark_complete` — the engine already knows the iteration is over.
132 */
133 class HegelStopTest : public std::exception {
134 public:
135 const char* what() const noexcept override {
136 return "test case stopped by backend";
